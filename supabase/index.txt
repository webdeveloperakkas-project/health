/*
 * Global Scope 24 — Publish Post Edge Function
 *
 * Copy this file into Supabase Edge Functions > publish-post > index.ts.
 *
 * Required Supabase Edge Function secrets:
 *   GITHUB_TOKEN
 *   GITHUB_REPOSITORY = webdeveloperakkas-project/health
 *   GITHUB_BRANCH = main
 *   SITE_URL = https://webdeveloperakkas-project.github.io/health
 *
 * SUPABASE_URL and SUPABASE_ANON_KEY are provided by Supabase.
 * SUPABASE_SERVICE_ROLE_KEY is provided by the Supabase Edge Runtime.
 *
 * Requests:
 *   POST { "action": "publish", "postIds": ["post-id-1", "post-id-2"] }
 *   POST { "action": "update_sitemap", "postIds": ["post-id-1"], "include": true }
 *
 * Only authenticated admins in public.admin_users can use this function.
 * Publishing updates Supabase status and removes legacy standalone article files.
 * Article routes are served by the main site through the repository's 404.html fallback.
 * Sitemap approval can add or remove selected already-published URLs.
 */

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
};

const CATEGORY_LABELS: Record<string, string> = {
  world: "World News",
  tips: "Daily Tips",
  technology: "Science & Technology",
  sports: "Sports",
  health: "Health",
  biography: "Biography",
  entertainment: "Entertainment",
  travel: "Travel",
  finance: "Finance",
  lifestyle: "Lifestyle",
};

function jsonResponse(status: number, body: Record<string, unknown>) {
  return new Response(JSON.stringify(body), {
    status,
    headers: { ...corsHeaders, "Content-Type": "application/json; charset=utf-8" },
  });
}

/* Match the title/category slug rules already used by script.js and admin.js. */
function makeSlug(value: unknown): string {
  return String(value ?? "")
    .normalize("NFC")
    .toLowerCase()
    .trim()
    .replace(/[^\p{L}\p{M}\p{N}]+/gu, "-")
    .replace(/^-+|-+$/g, "") || "article";
}

function postPath(post: Record<string, any>): string {
  const categoryKey = String(post.category || "");
  const categoryLabel = CATEGORY_LABELS[categoryKey];
  if (!categoryLabel) {
    throw new Error("Unsupported or inactive category on post " + String(post.id));
  }

  const stableCategoryPath = categoryKey === "technology" ? "technology" : makeSlug(categoryLabel);
  const parts = [stableCategoryPath];
  if (String(post.subcategory || "").trim()) {
    parts.push(makeSlug(post.subcategory));
  }
  parts.push(makeSlug(post.title));
  return parts.join("/");
}

async function supabaseRequest(path: string, serviceKey: string, method = "GET", body?: unknown) {
  const url = Deno.env.get("SUPABASE_URL") + "/rest/v1/" + path;
  const response = await fetch(url, {
    method,
    headers: {
      "apikey": serviceKey,
      "Authorization": "Bearer " + serviceKey,
      "Content-Type": "application/json",
      "Prefer": method === "PATCH" ? "return=representation" : "return=minimal",
    },
    body: body === undefined ? undefined : JSON.stringify(body),
  });
  const text = await response.text();
  if (!response.ok) throw new Error("Supabase request failed (" + response.status + "): " + text);
  return text ? JSON.parse(text) : null;
}

async function githubRequest(path: string, token: string, method = "GET", body?: unknown) {
  const response = await fetch("https://api.github.com" + path, {
    method,
    headers: {
      "Accept": "application/vnd.github+json",
      "Authorization": "Bearer " + token,
      "X-GitHub-Api-Version": "2022-11-28",
      "Content-Type": "application/json",
    },
    body: body === undefined ? undefined : JSON.stringify(body),
  });
  const text = await response.text();
  if (!response.ok) throw new Error("GitHub request failed (" + response.status + "): " + text);
  return text ? JSON.parse(text) : null;
}

async function githubPutFile(owner: string, repo: string, path: string, token: string, branch: string, content: string, message: string) {
  const encodedPath = path.split("/").map((part) => encodeURIComponent(part)).join("/");
  let existingSha: string | undefined;
  try {
    const existing = await githubRequest("/repos/" + owner + "/" + repo + "/contents/" + encodedPath + "?ref=" + encodeURIComponent(branch), token);
    existingSha = existing.sha;
  } catch (error) {
    if (!(error instanceof Error) || !error.message.includes("GitHub request failed (404)")) throw error;
  }
  const bytes = new TextEncoder().encode(content);
  let binary = "";
  bytes.forEach((byte) => { binary += String.fromCharCode(byte); });
  const body: Record<string, unknown> = { message, content: btoa(binary), branch };
  if (existingSha) body.sha = existingSha;
  return await githubRequest("/repos/" + owner + "/" + repo + "/contents/" + encodedPath, token, "PUT", body);
}

function xmlEscape(value: string): string {
  return value.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&apos;");
}

function buildSitemap(posts: Record<string, any>[], siteUrl: string): string {
  const entries = posts.map((post) => {
    const url = siteUrl + "/" + postPath(post) + "/";
    const modified = post.updated_at || post.published_at || post.created_at || new Date().toISOString();
    return "  <url><loc>" + xmlEscape(url) + "</loc><lastmod>" +
      xmlEscape(new Date(modified).toISOString().slice(0, 10)) + "</lastmod></url>";
  });
  return '<?xml version="1.0" encoding="UTF-8"?>\n' +
    '<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">\n' +
    '  <url><loc>' + xmlEscape(siteUrl + "/") + '</loc></url>\n' +
    entries.join("\n") + "\n</urlset>\n";
}

Deno.serve(async (request: Request) => {
  if (request.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }
  if (request.method !== "POST") {
    return jsonResponse(405, { error: "Method not allowed. Use POST." });
  }

  try {
    const supabaseUrl = Deno.env.get("SUPABASE_URL");
    const anonKey = Deno.env.get("SUPABASE_ANON_KEY");
    const serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");
    const githubToken = Deno.env.get("GITHUB_TOKEN");
    const repository = Deno.env.get("GITHUB_REPOSITORY") || "webdeveloperakkas-project/health";
    const branch = Deno.env.get("GITHUB_BRANCH") || "main";
    const siteUrl = (Deno.env.get("SITE_URL") || "https://webdeveloperakkas-project.github.io/health").replace(/\/+$/, "");

    if (!supabaseUrl || !anonKey || !serviceKey || !githubToken) {
      return jsonResponse(500, { error: "Missing required Edge Function environment configuration." });
    }

    const authHeader = request.headers.get("Authorization") || "";
    const accessToken = authHeader.replace(/^Bearer\s+/i, "").trim();
    if (!accessToken) return jsonResponse(401, { error: "Sign-in is required." });

    /* Verify the caller's Supabase access token, then check admin_users server-side. */
    const userResponse = await fetch(supabaseUrl + "/auth/v1/user", {
      headers: { "apikey": anonKey, "Authorization": "Bearer " + accessToken },
    });
    if (!userResponse.ok) return jsonResponse(401, { error: "Your session is invalid or expired. Please sign in again." });
    const user = await userResponse.json();

    const adminRows = await supabaseRequest(
      "admin_users?user_id=eq." + encodeURIComponent(user.id) + "&select=user_id&limit=1",
      serviceKey,
    );
    if (!Array.isArray(adminRows) || adminRows.length === 0) {
      return jsonResponse(403, { error: "Administrator permission is required." });
    }

    const payload = await request.json();
    const action = String(payload.action || "publish");
    const rawIds = Array.isArray(payload.postIds) ? payload.postIds : [];
    const postIds = [...new Set(rawIds.map((id: unknown) => String(id).trim()).filter(Boolean))];
    if (postIds.length === 0) return jsonResponse(400, { error: "Select at least one post." });
    if (postIds.length > 25) return jsonResponse(400, { error: "Process up to 25 posts per batch." });

    const filter = postIds.map((id) => '"' + id.replace(/["\\\\]/g, "") + '"').join(",");
    const selectedPosts = await supabaseRequest(
      "posts?id=in.(" + encodeURIComponent(filter) + ")&select=*&limit=25",
      serviceKey,
    );
    if (!Array.isArray(selectedPosts) || selectedPosts.length !== postIds.length) {
      return jsonResponse(404, { error: "One or more selected posts could not be found." });
    }

    const [owner, repo] = repository.split("/");
    if (!owner || !repo) throw new Error("GITHUB_REPOSITORY must use owner/repository format.");
    if (action === "publish") {
      for (const post of selectedPosts) {
        if (post.status !== "draft" && post.status !== "published") {
          return jsonResponse(409, { error: "Only Draft or already Published posts can be published. Check: " + post.title });
        }
        if (!String(post.title || "").trim() || !String(post.content_html || "").trim()) {
          return jsonResponse(422, { error: "A selected post is missing its title or article content." });
        }
        postPath(post);
      }

      const publishTime = new Date().toISOString();
      // Keep robots.txt current using the GitHub Contents API. Published article
      // content stays in Supabase and is rendered by the shared site route handler.
      // This removes the failing Git Trees operation and legacy HTML deletion.
      await githubPutFile(
        owner, repo, "robots.txt", githubToken, branch,
        "User-agent: *\nAllow: /\nSitemap: " + siteUrl + "/sitemap.xml\n",
        "Update Global Scope 24 robots.txt",
      );

      const draftPosts = selectedPosts.filter((post: Record<string, any>) => post.status === "draft");
      const draftIds = draftPosts.map((post: Record<string, any>) => '"' + String(post.id).replace(/["\\\\]/g, "") + '"').join(",");
      if (draftIds) {
        await supabaseRequest(
          "posts?id=in.(" + encodeURIComponent(draftIds) + ")",
          serviceKey,
          "PATCH",
          { status: "published", published_at: publishTime, updated_at: publishTime },
        );
      }
      return jsonResponse(200, {
        success: true,
        action: "publish",
        publishedCount: selectedPosts.length,
        published: selectedPosts.map((post: Record<string, any>) => ({
          id: post.id, title: post.title, path: postPath(post) + "/",
          url: siteUrl + "/" + postPath(post) + "/",
        })),
        message: "Posts published to Supabase. Article URLs use the shared site route handler; robots.txt was updated through the GitHub Contents API.",
      });
    }

    if (action === "update_sitemap") {
      const include = payload.include === true;
      for (const post of selectedPosts) {
        if (post.status !== "published") {
          return jsonResponse(409, { error: "Only published posts can be submitted to the sitemap: " + post.title });
        }
        postPath(post);
      }

      const idFilter = postIds.map((id) => '"' + id.replace(/["\\\\]/g, "") + '"').join(",");
      await supabaseRequest(
        "posts?id=in.(" + encodeURIComponent(idFilter) + ")",
        serviceKey,
        "PATCH",
        { sitemap_submitted: include, sitemap_submitted_at: include ? new Date().toISOString() : null },
      );

      try {
        const approvedPosts = await supabaseRequest(
          "posts?status=eq.published&sitemap_submitted=eq.true&select=id,title,category,subcategory,published_at,updated_at,created_at&order=published_at.desc&limit=5000",
          serviceKey,
        );
        await githubPutFile(
          owner, repo, "sitemap.xml", githubToken, branch,
          buildSitemap(Array.isArray(approvedPosts) ? approvedPosts : [], siteUrl),
          (include ? "Add approved URLs to" : "Remove URLs from") + " Global Scope 24 sitemap",
        );
        await githubPutFile(
          owner, repo, "robots.txt", githubToken, branch,
          "User-agent: *\nAllow: /\nSitemap: " + siteUrl + "/sitemap.xml\n",
          "Update Global Scope 24 robots.txt",
        );
        return jsonResponse(200, {
          success: true, action: "update_sitemap", include,
          submittedCount: selectedPosts.length,
          sitemapUrl: siteUrl + "/sitemap.xml",
          approvedUrlCount: Array.isArray(approvedPosts) ? approvedPosts.length : 0,
          message: "Sitemap and robots.txt updated through the GitHub Contents API. Search engines decide when to crawl and index approved URLs.",
        });
      } catch (sitemapError) {
        for (const post of selectedPosts) {
          await supabaseRequest(
            "posts?id=eq." + encodeURIComponent(String(post.id)),
            serviceKey,
            "PATCH",
            { sitemap_submitted: !!post.sitemap_submitted, sitemap_submitted_at: post.sitemap_submitted_at || null },
          );
        }
        throw sitemapError;
      }
    }

    return jsonResponse(400, { error: "Unsupported action. Use publish or update_sitemap." });
  } catch (error) {
    console.error("publish-post failed:", error);
    return jsonResponse(500, { error: error instanceof Error ? error.message : "Unexpected publishing error." });
  }
});
